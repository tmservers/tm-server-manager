/**
 *	Driver mode
 *
 *	Generic mode without any mode logic. It reports the match flow through typed
 *	Driver.* callbacks and waits on decision callbacks until the driver
 *	(tm-wasm-driver in the tm-server-bridge) acknowledged them with Driver.Ack.
 *	Points, eliminations and end conditions are all decided by the driver.
 *
 *	The callbacks and commands mirror tm_server_types::event::DriverEvent and
 *	tm_server_types::method::DriverCommand. Keep them in sync.
 */

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
// Functions
// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
Integer Driver_NextSeq() {
	G_Driver_Seq += 1;
	return G_Driver_Seq;
}

K_Driver_Player Driver_Player(CSmPlayer _Player) {
	return K_Driver_Player {
		account_id = _Player.User.WebServicesUserId,
		name = _Player.User.Name,
		// Clan 0 is no team, 1 is blue and 2 is red.
		team = _Player.CurrentClan - 1,
		spectator = Spectators.exists(_Player)
	};
}

K_Driver_Player[] Driver_Players() {
	declare K_Driver_Player[] DriverPlayers;
	foreach (Player in Players) {
		if (Player.User == Null) continue;
		DriverPlayers.add(Driver_Player(Player));
	}
	return DriverPlayers;
}

Boolean Driver_CanSpawn(CSmPlayer _Player) {
	if (_Player.User == Null) return False;
	declare Text AccountId = _Player.User.WebServicesUserId;
	return !G_Driver_CanSpawn.existskey(AccountId) || G_Driver_CanSpawn[AccountId];
}

/// Time attack rounds rank by the best time, everything else by points.
Void Driver_SortScores() {
	if (G_Driver_RoundIsTimeAttack) {
		Race::SortScores(Race::C_Sort_BestRaceTime);
	} else {
		Race::SortScores(Race::C_Sort_TotalPoints);
	}
}

CSmScore Driver_FindScore(Text _AccountId) {
	foreach (Score in Scores) {
		if (Score.User != Null && Score.User.WebServicesUserId == _AccountId) return Score;
	}
	return Null;
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Sends a decision callback and blocks until the driver acknowledged it.
 *	Commands keep being applied while waiting. The callback is sent again
 *	periodically because the driver answers repeated callbacks without
 *	applying them twice.
 *
 *	@param	_Callback		Name of the callback
 *	@param	_Seq				Seq of the callback which gets acknowledged
 *	@param	_Body				Json payload of the callback
 */
Void Driver_Decide(Text _Callback, Integer _Seq, Text _Body) {
	G_Driver_PendingSeq = _Seq;
	XmlRpc::SendCallback(_Callback, [_Body]);

	declare Integer ResendTime = Now + S_DriverDecisionTimeout;
	declare Boolean Waiting = False;
	while (G_Driver_PendingSeq == _Seq && !ServerShutdownRequested) {
		MB_Yield();
		if (G_Driver_PendingSeq == _Seq && Now >= ResendTime) {
			if (!Waiting) {
				Waiting = True;
				UIModules_BigMessage::SetMessage(_("Waiting for the match driver..."));
			}
			XmlRpc::SendCallback(_Callback, [_Body]);
			ResendTime = Now + C_Driver_ResendInterval;
		}
	}
	if (Waiting) UIModules_BigMessage::SetMessage(G_Driver_Message);
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Computes the end time of the round after the first player finished.
Integer Driver_FinishTimeout() {
	if (G_Driver_FinishTimeout == "off") return -1;
	if (G_Driver_FinishTimeout != "auto") return Now + TL::ToInteger(G_Driver_FinishTimeout);

	declare Integer FinishTimeout = 5000;
	if (Map.TMObjective_IsLapRace && Race::GetLapsNb() > 0 && Map.TMObjective_NbLaps > 0) {
		FinishTimeout += ((Map.TMObjective_AuthorTime / Map.TMObjective_NbLaps) * Race::GetLapsNb()) / 6;
	} else {
		FinishTimeout += Map.TMObjective_AuthorTime / 6;
	}
	return Now + FinishTimeout;
}

Void Driver_SetLayer(Text _Name, Text _Manialink) {
	if (G_Driver_Layers.existskey(_Name) && UIManager.UILayers.existskey(G_Driver_Layers[_Name])) {
		UIManager.UILayers[G_Driver_Layers[_Name]].ManialinkPage = _Manialink;
		return;
	}
	declare CUILayer Layer <=> UIManager.UILayerCreate();
	Layer.ManialinkPage = _Manialink;
	G_Driver_Layers[_Name] = Layer.Id;
	UIManager.UIAll.UILayers.add(Layer);
}

Void Driver_RemoveLayer(Text _Name) {
	if (!G_Driver_Layers.existskey(_Name)) return;
	declare Ident LayerId = G_Driver_Layers[_Name];
	declare Boolean Forgotten = G_Driver_Layers.removekey(_Name);
	declare Boolean Detached = UIManager.UIAll.UILayers.removekey(LayerId);
	if (UIManager.UILayers.existskey(LayerId)) UIManager.UILayerDestroy(UIManager.UILayers[LayerId]);
}

/** Applies a command sent by the driver.
 *
 *	@param	_Method			Name of the command (Driver.*)
 *	@param	_Params			Text params as encoded by DriverCommand::params
 */
Void Driver_ApplyCommand(Text _Method, Text[] _Params) {
	if (_Method == C_Method_Ack) {
		if (_Params.count >= 1 && TL::ToInteger(_Params[0]) == G_Driver_PendingSeq) {
			G_Driver_PendingSeq = 0;
		}
	} else if (_Method == C_Method_SetPlayerPoints) {
		if (_Params.count < 4) return;
		declare CSmScore Score <=> Driver_FindScore(_Params[0]);
		if (Score == Null) return;
		if (_Params[1] != "") Scores::SetPlayerRoundPoints(Score, TL::ToInteger(_Params[1]));
		if (_Params[2] != "") Scores::SetPlayerMapPoints(Score, TL::ToInteger(_Params[2]));
		if (_Params[3] != "") Scores::SetPlayerMatchPoints(Score, TL::ToInteger(_Params[3]));
	} else if (_Method == C_Method_SetCanSpawn) {
		if (_Params.count < 2) return;
		G_Driver_CanSpawn[_Params[0]] = (_Params[1] == "1");
	} else if (_Method == C_Method_SetFinishTimeout) {
		if (_Params.count < 1) return;
		G_Driver_FinishTimeout = _Params[0];
	} else if (_Method == C_Method_ForceEndRound) {
		G_Driver_ForceEndRound = True;
	} else if (_Method == C_Method_EndMap) {
		G_Driver_EndMap = True;
	} else if (_Method == C_Method_EndMatch) {
		G_Driver_EndMatch = True;
	} else if (_Method == C_Method_SetMessage) {
		if (_Params.count < 1) return;
		G_Driver_Message = _Params[0];
		UIModules_BigMessage::SetMessage(G_Driver_Message);
	} else if (_Method == C_Method_SetLayer) {
		if (_Params.count < 2) return;
		Driver_SetLayer(_Params[0], _Params[1]);
	} else if (_Method == C_Method_RemoveLayer) {
		if (_Params.count < 1) return;
		Driver_RemoveLayer(_Params[0]);
	} else if (_Method == C_Method_SetNetState) {
		if (_Params.count < 1) return;
		declare netwrite Text Net_Driver_State for Teams[0] = "";
		declare netwrite Integer Net_Driver_StateSerial for Teams[0] = 0;
		Net_Driver_State = _Params[0];
		Net_Driver_StateSerial += 1;
	} else if (_Method == C_Method_SetRoundFormat) {
		if (_Params.count < 1) return;
		if (_Params[0] == "time_attack" && _Params.count >= 2) {
			G_Driver_TimeAttackLimit = TL::ToInteger(_Params[1]);
		} else {
			G_Driver_TimeAttackLimit = 0;
		}
	}
}
