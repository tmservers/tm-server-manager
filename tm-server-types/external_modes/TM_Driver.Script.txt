/**
 *	Driver mode
 *
 *	Generic mode without any mode logic. The match flow reaches the driver
 *	(tm-mode-driver, run by its runner until tm-server-bridge integrates it)
 *	through the standard callbacks of the base script. At a few points the
 *	script waits with the Driver.Hold callback until the driver answers with
 *	Driver.Continue.
 *
 *	The script keeps no points. The driver keeps them in standings every mode
 *	shares and publishes them for the manialinks together with the round, live while
 *	it runs, and the players it knows. The script writes them into typed netvars
 *	(tm_server_types::method::DRIVER_NETVARS), players by their index. Race times
 *	reach the driver through the standard waypoint callbacks.
 *
 *	Only what is not available through the standard XML-RPC methods is added
 *	here. It mirrors tm_server_types::event::DriverHold and
 *	tm_server_types::method::DriverCommand. Keep them in sync.
 */

	declare Text[] Driver_UiEvent = [Driver_AccountId, Event.CustomEventType];

Boolean Driver_CanSpawn(CSmPlayer _Player) {
	if (_Player.User == Null) return False;
	declare Boolean[Text] Driver_SpawnPermissions for This = [];
	declare Text AccountId = _Player.User.WebServicesUserId;
	return !Driver_SpawnPermissions.existskey(AccountId) || Driver_SpawnPermissions[AccountId];
}

/** Applies the laps and give ups the driver set to the round that starts.
 *	Only what the driver asked for changes, the rest stays as the base script set
 *	it, and all of it comes back once the driver wants the defaults again.
 */
Void Driver_ApplyRaceRules() {
	// What the driver leaves open the config decides, as for the modes of Nadeo: its laps,
	// otherwise the ones of the map, and its respawn behaviour, otherwise the normal one.
	if (G_Driver_InfiniteLaps) {
		Race::SetLapsSettings(True, -1);
	} else if (G_Driver_LapsNb > 0) {
		Race::SetLapsSettings(False, G_Driver_LapsNb);
	} else {
		Race::SetLapsSettings(S_InfiniteLaps, S_ForceLapsNb);
	}
	declare Integer Respawn = G_Driver_Respawn;
	if (Respawn == 0) Respawn = S_RespawnBehaviour;
	if (Respawn == 0) Respawn = Race::C_RespawnBehaviour_Normal;
	Race::SetRespawnBehaviour(Respawn);
}

/// Time attack rounds rank by the best race, everything else by the last race.
Void Driver_SortScores() {
	if (G_Driver_RoundIsTimeAttack) {
		Race::SortScores(Race::C_Sort_BestRaceTime);
	} else {
		Race::SortScores(Race::C_Sort_PrevRaceTime);
	}
}

Void Driver_SendHold(K_Driver_Hold _Hold) {
	XmlRpc::SendCallback(C_Callback_Hold, [_Hold.tojson()]);
}

/// Sends on what players did in the layers of the driver, Match_Yield queued it. Events
/// of other layers are none of the driver's business.
Void Driver_SendUiEvents() {
	declare Ident[] Driver_PendingUiLayers for This = [];
	declare Text[][] Driver_PendingUiEvents for This = [];
	foreach (Index => Queued in Driver_PendingUiEvents) {
		if (!Driver_PendingUiLayers.existskey(Index) || Queued.count < 2) continue;
		declare K_Driver_UiEvent UiEvent;
		foreach (Name => LayerId in G_Driver_Layers) {
			if (LayerId == Driver_PendingUiLayers[Index]) UiEvent.layer = Name;
		}
		if (UiEvent.layer == "") continue;
		UiEvent.account_id = Queued[0];
		UiEvent.name = Queued[1];
		foreach (Position => Arg in Queued) {
			if (Position > 1) UiEvent.data.add(Arg);
		}
		XmlRpc::SendCallback(C_Callback_UiEvent, [UiEvent.tojson()]);
	}
	Driver_PendingUiLayers.clear();
	Driver_PendingUiEvents.clear();
}

/// The index of a player in the netvars of the players, -1 for one the driver did not publish.
Integer Driver_UserIndex(Text _AccountId) {
	declare netwrite Text[Integer] Net_Driver_UserAccountIds for Teams[0] = [];
	foreach (Index => AccountId in Net_Driver_UserAccountIds) {
		if (AccountId == _AccountId) return Index;
	}
	return -1;
}

/// The team of a player, -1 without one.
Integer Driver_TeamOf(Integer _Index) {
	declare netwrite Integer[Integer] Net_Driver_UserTeams for Teams[0] = [];
	if (Net_Driver_UserTeams.existskey(_Index)) return Net_Driver_UserTeams[_Index];
	return -1;
}

/// Whether a player is one of an audience.
Boolean Driver_InAudience(CSmPlayer _Player, K_Driver_Audience _Audience) {
	if (_Audience.kind == C_Audience_Everyone) return True;
	if (_Player.User == Null) return False;
	switch (_Audience.kind) {
		case C_Audience_Players: return !_Player.RequestsSpectate;
		case C_Audience_Spectators: return _Player.RequestsSpectate;
		case C_Audience_Accounts: return _Audience.accounts.exists(_Player.User.WebServicesUserId);
		case C_Audience_Teams: {
			declare Integer Team = Driver_TeamOf(Driver_UserIndex(_Player.User.WebServicesUserId));
			return Team >= 0 && _Audience.teams.exists(Team);
		}
	}
	return False;
}

/// The text a marker on a player shows.
Text Driver_MarkerLabel(Text _Label, Integer _Index) {
	if (_Label == "" || _Index < 0) return "";
	declare netwrite Text[Integer] Net_Driver_UserNames for Teams[0] = [];
	declare netwrite Text[Integer] Net_Driver_UserClubTags for Teams[0] = [];
	declare netwrite Text[Integer] Net_Driver_TeamShortNames for Teams[0] = [];
	declare netwrite Integer[] Net_Driver_StandingsPlayers for Teams[0] = [];
	declare netwrite Integer[] Net_Driver_StandingsPositions for Teams[0] = [];
	declare netwrite Integer[] Net_Driver_StandingsScores for Teams[0] = [];
	if (_Label == C_MarkerLabel_Name && Net_Driver_UserNames.existskey(_Index)) {
		return Net_Driver_UserNames[_Index];
	}
	if (_Label == C_MarkerLabel_ClubTag && Net_Driver_UserClubTags.existskey(_Index)) {
		return Net_Driver_UserClubTags[_Index];
	}
	if (_Label == C_MarkerLabel_TeamShortName) {
		declare Integer Team = Driver_TeamOf(_Index);
		if (Net_Driver_TeamShortNames.existskey(Team)) return Net_Driver_TeamShortNames[Team];
		return "";
	}
	foreach (Entry => Player in Net_Driver_StandingsPlayers) {
		if (Player != _Index) continue;
		if (_Label == C_MarkerLabel_Score) return TL::ToText(Net_Driver_StandingsScores[Entry]);
		if (_Label == C_MarkerLabel_Position && Net_Driver_StandingsPositions[Entry] > 0) {
			return TL::ToText(Net_Driver_StandingsPositions[Entry]);
		}
	}
	return "";
}

Void Driver_SetupMarker(CUIConfigMarker _Marker, K_Driver_Marker _Binding, Integer _Index) {
	if (_Marker == Null) return;
	_Marker.ManialinkFrameId = _Binding.frame_id;
	_Marker.Label = Driver_MarkerLabel(_Binding.label, _Index);
	_Marker.HideOnSelf = _Binding.hide_on_self;
	if (_Binding.max_distance > 0) _Marker.DistMax = _Binding.max_distance * 1.;
	switch (_Binding.visibility) {
		case C_MarkerVisibility_WhenInFrustum: _Marker.HudVisibility = CUIConfigMarker::EHudVisibility::WhenInFrustum;
		case C_MarkerVisibility_WhenVisible: _Marker.HudVisibility = CUIConfigMarker::EHudVisibility::WhenVisible;
		case C_MarkerVisibility_WhenInMiddleOfScreen: _Marker.HudVisibility = CUIConfigMarker::EHudVisibility::WhenInMiddleOfScreen;
		default: _Marker.HudVisibility = CUIConfigMarker::EHudVisibility::Always;
	}
}

/** Builds the markers of every markers layer again once the players, their standings or
 *	the layers changed. The driver owns the markers of UIAll.
 */
Void Driver_SyncMarkers() {
	declare netwrite Integer Net_Driver_StandingsSerial for Teams[0] = 0;
	declare netwrite Integer Net_Driver_UsersSerial for Teams[0] = 0;
	declare netwrite Integer Net_Driver_TeamsSerial for Teams[0] = 0;
	declare Text From = TL::ToText(Net_Driver_StandingsSerial)^"/"^Net_Driver_UsersSerial^"/"^Net_Driver_TeamsSerial^"/"^Map.Id;
	foreach (Player in AllPlayers) {
		if (Player.User != Null) From ^= "/"^Player.User.WebServicesUserId^Player.RequestsSpectate;
	}
	if (From == G_Driver_MarkersBuiltFrom) return;
	G_Driver_MarkersBuiltFrom = From;

	UIManager.UIAll.ClearMarkers();
	foreach (Name => Markers in G_Driver_LayerMarkers) {
		foreach (Binding in Markers) {
			if (Binding.anchor == C_MarkerAnchor_Players) {
				foreach (Player in AllPlayers) {
					if (Player.User == Null || !Driver_InAudience(Player, Binding.players)) continue;
					declare Integer Index = Driver_UserIndex(Player.User.WebServicesUserId);
					Driver_SetupMarker(UIManager.UIAll.AddMarker(Player), Binding, Index);
				}
			} else if (Binding.anchor == C_MarkerAnchor_Position) {
				if (Binding.position.count != 3) continue;
				declare Vec3 Position = <Binding.position[0] * 1., Binding.position[1] * 1., Binding.position[2] * 1.>;
				Driver_SetupMarker(UIManager.UIAll.AddMarker(Position), Binding, -1);
			} else {
				foreach (Landmark in MapLandmarks) {
					if (Landmark.Tag == Binding.anchor) {
						Driver_SetupMarker(UIManager.UIAll.AddMarker(Landmark), Binding, -1);
					}
				}
			}
		}
	}
}

/// The values of the team of a player and their own, theirs win.
Void Driver_SyncMyData(CUIConfig UI, Text _AccountId, Integer _Team) {
	declare Text Driver_MyDataFrom for UI = "";
	declare Text From = G_Driver_PlayerDataSerial^":"^_Team;
	if (Driver_MyDataFrom == From) return;
	Driver_MyDataFrom = From;
	declare Text[Text] Data;
	if (G_Driver_TeamData.existskey(_Team)) Data = G_Driver_TeamData[_Team];
	if (G_Driver_AccountData.existskey(_AccountId)) {
		foreach (Key => Value in G_Driver_AccountData[_AccountId]) {
			Data[Key] = Value;
		}
	}
	declare netwrite Integer Net_Driver_MyDataSerial for UI = 0;
	declare netwrite Text[Text] Net_Driver_MyData for UI = [];
	Net_Driver_MyData = Data;
	Net_Driver_MyDataSerial += 1;
}

/** Brings the layers of every player up to date with their audiences, at most once a
 *	second: players join, leave, start spectating or play for a team. Every player also
 *	learns their index and team for the manialinks, `for UI`.
 */
Void Driver_SyncAudiences() {
	if (G_Driver_AudiencesSynced >= 0 && Now < G_Driver_AudiencesSynced + C_AudienceSyncInterval) return;
	G_Driver_AudiencesSynced = Now;

	foreach (Player in AllPlayers) {
		declare CUIConfig UI <=> UIManager.GetUI(Player);
		if (UI == Null || Player.User == Null) continue;
		declare Integer Index = Driver_UserIndex(Player.User.WebServicesUserId);
		declare netwrite Integer Net_Driver_MyUser for UI = -1;
		declare netwrite Integer Net_Driver_MyTeam for UI = -1;
		Net_Driver_MyUser = Index;
		Net_Driver_MyTeam = Driver_TeamOf(Index);
		Driver_SyncMyData(UI, Player.User.WebServicesUserId, Net_Driver_MyTeam);

		foreach (Name => LayerId in G_Driver_Layers) {
			if (!UIManager.UILayers.existskey(LayerId) || !G_Driver_LayerAudiences.existskey(Name)) continue;
			declare K_Driver_Audience Audience = G_Driver_LayerAudiences[Name];
			if (Audience.kind == C_Audience_Everyone) continue;
			declare Boolean Has = UI.UILayers.existskey(LayerId);
			declare Boolean Wants = Driver_InAudience(Player, Audience);
			if (Wants && !Has) UI.UILayers.add(UIManager.UILayers[LayerId]);
			if (!Wants && Has) {
				declare Boolean Removed = UI.UILayers.removekey(LayerId);
			}
		}
	}
	Driver_SyncMarkers();
}

/** Creates or replaces a layer.
 *	The kind says where the game shows it: scores_table is the scores table, which the
 *	game shows while a player holds the scores table key and whenever
 *	ScoreTableVisibility forces it. A layer for everyone is a layer of UIAll, any
 *	other audience a layer of the UI of every player in it, see Driver_SyncAudiences.
 */
Void Driver_SetLayer(Text _Name, Text _Manialink, Text _Kind, Text _AttachId, Text _Audience, Text _Markers) {
	declare CUILayer Layer <=> Null;
	if (G_Driver_Layers.existskey(_Name) && UIManager.UILayers.existskey(G_Driver_Layers[_Name])) {
		Layer <=> UIManager.UILayers[G_Driver_Layers[_Name]];
	}
	declare Boolean IsNew = Layer == Null;
	if (IsNew) Layer <=> UIManager.UILayerCreate();

	switch (_Kind) {
		case C_LayerKind_ScoresTable: Layer.Type = CUILayer::EUILayerType::ScoresTable;
		case C_LayerKind_ScreenIn3d: Layer.Type = CUILayer::EUILayerType::ScreenIn3d;
		case C_LayerKind_Markers: Layer.Type = CUILayer::EUILayerType::Markers;
		case C_LayerKind_AltMenu: Layer.Type = CUILayer::EUILayerType::AltMenu;
		case C_LayerKind_CutScene: Layer.Type = CUILayer::EUILayerType::CutScene;
		case C_LayerKind_LoadingScreen: Layer.Type = CUILayer::EUILayerType::LoadingScreen;
		default: Layer.Type = CUILayer::EUILayerType::Normal;
	}
	Layer.AttachId = _AttachId;
	Layer.ManialinkPage = _Manialink;

	declare K_Driver_Audience Audience;
	Audience.kind = C_Audience_Everyone;
	if (_Audience != "" && !Audience.fromjson(_Audience)) Audience.kind = C_Audience_Everyone;
	declare K_Driver_Marker[] Markers;
	if (_Markers != "" && !Markers.fromjson(_Markers)) Markers = [];

	if (IsNew) G_Driver_Layers[_Name] = Layer.Id;
	declare Boolean Shared = UIManager.UIAll.UILayers.existskey(Layer.Id);
	if (Audience.kind == C_Audience_Everyone && !Shared) {
		UIManager.UIAll.UILayers.add(Layer);
	} else if (Audience.kind != C_Audience_Everyone && Shared) {
		declare Boolean Detached = UIManager.UIAll.UILayers.removekey(Layer.Id);
	}
	G_Driver_LayerAudiences[_Name] = Audience;
	G_Driver_LayerMarkers[_Name] = Markers;
	// The players get it right away.
	G_Driver_AudiencesSynced = -1;
	G_Driver_MarkersBuiltFrom = "";
	Driver_SyncAudiences();
}

Void Driver_RemoveLayer(Text _Name) {
	if (!G_Driver_Layers.existskey(_Name)) return;
	declare Ident LayerId = G_Driver_Layers[_Name];
	declare Boolean Forgotten = G_Driver_Layers.removekey(_Name);
	declare Boolean NoAudience = G_Driver_LayerAudiences.removekey(_Name);
	declare Boolean NoMarkers = G_Driver_LayerMarkers.removekey(_Name);
	declare Boolean Detached = UIManager.UIAll.UILayers.removekey(LayerId);
	foreach (Player in AllPlayers) {
		declare CUIConfig UI <=> UIManager.GetUI(Player);
		if (UI != Null && UI.UILayers.existskey(LayerId)) {
			declare Boolean Removed = UI.UILayers.removekey(LayerId);
		}
	}
	if (UIManager.UILayers.existskey(LayerId)) UIManager.UILayerDestroy(UIManager.UILayers[LayerId]);
	G_Driver_MarkersBuiltFrom = "";
}

/// Every column has a value for every entry.
Boolean Driver_ValidEntries(K_Driver_Entries _Entries) {
	declare Integer Count = _Entries.players.count;
	return _Entries.positions.count == Count
		&& _Entries.scores.count == Count
		&& _Entries.times.count == Count
		&& _Entries.statuses.count == Count
		&& _Entries.labels.count == _Entries.labeled.count
		&& _Entries.label_tones.count == _Entries.labeled.count;
}

/// Every column has a value for every team.
Boolean Driver_ValidTeamEntries(K_Driver_TeamEntries _Teams) {
	declare Integer Count = _Teams.teams.count;

		&& _Teams.scores.count == Count
		&& _Teams.map_scores.count == Count
		&& _Teams.labels.count == _Teams.labeled.count
		&& _Teams.label_tones.count == _Teams.labeled.count;
}

/// The values with the keys replaced, an empty value removes its key.
Text[Text] Driver_MergeData(Text[Text] _Data, K_Driver_UiData _Change) {
	declare Text[Text] Data = _Data;
	foreach (Index => Key in _Change.keys) {
		if (_Change.values[Index] == "") {
			declare Boolean Removed = Data.removekey(Key);
		} else {
			Data[Key] = _Change.values[Index];
		}
	}
	return Data;
}

/// Writes values for the manialinks: for everyone into Net_Driver_Data, for a team or
/// a player into Net_Driver_MyData of every player they concern.

	declare K_Driver_UiData Change;
	if (!Change.fromjson(_Json) || Change.keys.count != Change.values.count) return;
	if (Change.scope == C_UiData_All) {
		declare netwrite Text[Text] Net_Driver_Data for Teams[0] = [];
		declare netwrite Integer Net_Driver_DataSerial for Teams[0] = 0;
		Net_Driver_Data = Driver_MergeData(Net_Driver_Data, Change);
		Net_Driver_DataSerial += 1;
		return;
	}
	if (Change.scope == C_UiData_Team) {
		declare Text[Text] Data;
		if (G_Driver_TeamData.existskey(Change.team)) Data = G_Driver_TeamData[Change.team];

		if (G_Driver_AccountData.existskey(Change.account)) Data = G_Driver_AccountData[Change.account];
		G_Driver_AccountData[Change.account] = Driver_MergeData(Data, Change);
	} else {
		return;
	}
	G_Driver_PlayerDataSerial += 1;
	// The players get it right away.
	G_Driver_AudiencesSynced = -1;
}

/** Applies a command sent by the driver.
 *
 *	@param	_Method			Name of the command (Driver.*)
 *	@param	_Params			Text params as encoded by DriverCommand::call
 */
Void Driver_ApplyCommand(Text _Method, Text[] _Params) {
	if (_Method == C_Method_Continue) {
		if (_Params.count >= 1 && TL::ToInteger(_Params[0]) == G_Driver_PendingSeq) {
			G_Driver_PendingSeq = 0;
		}
	} else if (_Method == C_Method_SetRoundFormat) {
		if (_Params.count < 2) return;
		if (_Params[0] == "time_attack") {
			G_Driver_TimeAttackLimit = TL::ToInteger(_Params[1]);
		} else {
			G_Driver_TimeAttackLimit = 0;
			G_Driver_FinishTimeout = _Params[1];
		}
	} else if (_Method == C_Method_SetRaceRules) {
		if (_Params.count < 2) return;
		// map, infinite or a number of laps, then config or a C_RespawnBehaviour_* of
		// the race library.
		G_Driver_InfiniteLaps = _Params[0] == "infinite";
		G_Driver_LapsNb = 0;
		if (_Params[0] != "map" && !G_Driver_InfiniteLaps) {
			G_Driver_LapsNb = ML::Max(TL::ToInteger(_Params[0]), 0);
		}
		G_Driver_Respawn = 0;
		if (_Params[1] != "config") {
			G_Driver_Respawn = ML::Clamp(TL::ToInteger(_Params[1]), Race::C_RespawnBehaviour_Normal, Race::C_RespawnBehaviour_NeverGiveUp);
		}
	} else if (_Method == C_Method_SetCanSpawn) {
		if (_Params.count < 2) return;
		declare Boolean[Text] Driver_SpawnPermissions for This = [];
		Driver_SpawnPermissions[_Params[0]] = (_Params[1] == "1");
	} else if (_Method == C_Method_EndMatch) {
		G_Driver_EndMatch = True;
	} else if (_Method == C_Method_SetLayer) {
		if (_Params.count < 2) return;
		declare Text[] Rest = ["normal", "", "", ""];
		for (Index, 2, 5) {
			if (_Params.count > Index) Rest[Index - 2] = _Params[Index];
		}
		Driver_SetLayer(_Params[0], _Params[1], Rest[0], Rest[1], Rest[2], Rest[3]);
	} else if (_Method == C_Method_RemoveLayer) {
		if (_Params.count < 1) return;
		Driver_RemoveLayer(_Params[0]);
	} else if (_Method == C_Method_SetNetState) {
		if (_Params.count < 1) return;
		declare netwrite Text Net_Driver_State for Teams[0] = "";
		declare netwrite Integer Net_Driver_StateSerial for Teams[0] = 0;
		Net_Driver_State = _Params[0];
		Net_Driver_StateSerial += 1;
	} else if (_Method == C_Method_SetStandings) {
		if (_Params.count >= 1) Driver_SetStandings(_Params[0]);
	} else if (_Method == C_Method_SetRound) {
		if (_Params.count >= 1) Driver_SetRound(_Params[0]);
	} else if (_Method == C_Method_SetUsers) {
		if (_Params.count >= 1) Driver_SetUsers(_Params[0]);
	} else if (_Method == C_Method_SetTeams) {
		if (_Params.count >= 1) Driver_SetTeams(_Params[0]);
	} else if (_Method == C_Method_SetUiData) {
		if (_Params.count >= 1) Driver_SetUiData(_Params[0]);
	}
}

/// Applies the commands Match_Yield queued, in the order they arrived.
Void Driver_ApplyCommands() {
	declare Text[][] Driver_PendingCommands for This = [];
	foreach (Command in Driver_PendingCommands) {
		declare Text[] Params;
		foreach (Index => Param in Command) {
			if (Index > 0) Params.add(Param);
		}
		if (Command.count > 0) Driver_ApplyCommand(Command[0], Params);
	}
	Driver_PendingCommands.clear();
	// Players come and go between the commands.
	Driver_SyncAudiences();
	Driver_SendUiEvents();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Waits until the driver continues the hold.
 *	Commands keep being applied while waiting. The hold is sent again
 *	periodically because the driver answers repeated holds without
 *	deciding twice.
 *
 *	@param	_Section		Where the script waits (C_Section_*)
 *	@param	_Forced			RoundEnd only: the round should not count
 */
Void Driver_Hold(Text _Section, Boolean _Forced) {
	G_Driver_Seq += 1;
	declare K_Driver_Hold Hold = K_Driver_Hold {
		session = G_Driver_Session,
		seq = G_Driver_Seq,
		section = _Section,
		forced = _Forced
	};
	G_Driver_PendingSeq = Hold.seq;
	Driver_SendHold(Hold);

	declare Integer ResendTime = Now + S_DriverDecisionTimeout;
	declare Boolean Waiting = False;
	while (G_Driver_PendingSeq == Hold.seq && !ServerShutdownRequested) {
		MB_Yield();
		Driver_ApplyCommands();
		if (G_Driver_PendingSeq == Hold.seq && Now >= ResendTime) {
			if (!Waiting) {
				Waiting = True;
				UIModules_BigMessage::SetMessage(_("Waiting for the match driver..."));
			}
			Driver_SendHold(Hold);
			ResendTime = Now + C_Driver_ResendInterval;
		}
	}
	if (Waiting) UIModules_BigMessage::SetMessage("");
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Computes the end time of the round after the first player finished.
Integer Driver_FinishTimeout() {
	if (G_Driver_FinishTimeout == "off") return -1;
	if (G_Driver_FinishTimeout != "auto") return Now + TL::ToInteger(G_Driver_FinishTimeout);

	declare Integer FinishTimeout = 5000;
	if (Map.TMObjective_IsLapRace && Race::GetLapsNb() > 0 && Map.TMObjective_NbLaps > 0) {
		FinishTimeout += ((Map.TMObjective_AuthorTime / Map.TMObjective_NbLaps) * Race::GetLapsNb()) / 6;
	} else {
		FinishTimeout += Map.TMObjective_AuthorTime / 6;
	}
	return Now + FinishTimeout;
}
