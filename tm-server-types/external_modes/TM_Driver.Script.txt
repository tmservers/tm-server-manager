/**
 *	Driver mode
 *
 *	Generic mode without any mode logic. The match flow reaches the driver
 *	(tm-wasm-driver in the tm-server-bridge) through the standard callbacks of
 *	the base script. At a few points the script waits with the Driver.Hold
 *	callback until the driver answers with Driver.Continue.
 *
 *	The script keeps no points. The driver keeps them in standings every mode
 *	shares and publishes them for the manialinks through Net_Driver_Standings,
 *	together with the last round (Net_Driver_Round) and the players it knows
 *	(Net_Driver_Users). Race times reach the driver through the standard
 *	waypoint callbacks.
 *
 *	Only what is not available through the standard XML-RPC methods is added
 *	here. It mirrors tm_server_types::event::DriverHold and
 *	tm_server_types::method::DriverCommand. Keep them in sync.
 */

Boolean Driver_CanSpawn(CSmPlayer _Player) {
	if (_Player.User == Null) return False;
	declare Boolean[Text] Driver_SpawnPermissions for This = [];
	declare Text AccountId = _Player.User.WebServicesUserId;
	return !Driver_SpawnPermissions.existskey(AccountId) || Driver_SpawnPermissions[AccountId];
}

/// Time attack rounds rank by the best race, everything else by the last race.
Void Driver_SortScores() {
	if (G_Driver_RoundIsTimeAttack) {
		Race::SortScores(Race::C_Sort_BestRaceTime);
	} else {
		Race::SortScores(Race::C_Sort_PrevRaceTime);
	}
}

Void Driver_SendHold(K_Driver_Hold _Hold) {
	XmlRpc::SendCallback(C_Callback_Hold, [_Hold.tojson()]);
}

/** Creates or replaces a layer for every player.
 *	Layers of the kind scores_table are the scores table: the game shows them while
 *	a player holds the scores table key and whenever ScoreTableVisibility forces it.
 */
Void Driver_SetLayer(Text _Name, Text _Manialink, Text _Kind) {
	declare CUILayer Layer <=> Null;
	if (G_Driver_Layers.existskey(_Name) && UIManager.UILayers.existskey(G_Driver_Layers[_Name])) {
		Layer <=> UIManager.UILayers[G_Driver_Layers[_Name]];
	}
	declare Boolean IsNew = Layer == Null;
	if (IsNew) Layer <=> UIManager.UILayerCreate();

	if (_Kind == C_LayerKind_ScoresTable) {
		Layer.Type = CUILayer::EUILayerType::ScoresTable;
	} else {
		Layer.Type = CUILayer::EUILayerType::Normal;
	}
	Layer.ManialinkPage = _Manialink;

	if (IsNew) {
		G_Driver_Layers[_Name] = Layer.Id;
		UIManager.UIAll.UILayers.add(Layer);
	}
}

Void Driver_RemoveLayer(Text _Name) {
	if (!G_Driver_Layers.existskey(_Name)) return;
	declare Ident LayerId = G_Driver_Layers[_Name];
	declare Boolean Forgotten = G_Driver_Layers.removekey(_Name);
	declare Boolean Detached = UIManager.UIAll.UILayers.removekey(LayerId);
	if (UIManager.UILayers.existskey(LayerId)) UIManager.UILayerDestroy(UIManager.UILayers[LayerId]);
}

/** Applies a command sent by the driver.
 *
 *	@param	_Method			Name of the command (Driver.*)
 *	@param	_Params			Text params as encoded by DriverCommand::call
 */
Void Driver_ApplyCommand(Text _Method, Text[] _Params) {
	if (_Method == C_Method_Continue) {
		if (_Params.count >= 1 && TL::ToInteger(_Params[0]) == G_Driver_PendingSeq) {
			G_Driver_PendingSeq = 0;
		}
	} else if (_Method == C_Method_SetRoundFormat) {
		if (_Params.count < 2) return;
		if (_Params[0] == "time_attack") {
			G_Driver_TimeAttackLimit = TL::ToInteger(_Params[1]);
		} else {
			G_Driver_TimeAttackLimit = 0;
			G_Driver_FinishTimeout = _Params[1];
		}
	} else if (_Method == C_Method_SetWarmUp) {
		if (_Params.count < 3) return;
		G_Driver_WarmUpNb = TL::ToInteger(_Params[0]);
		G_Driver_WarmUpDuration = TL::ToInteger(_Params[1]);
		if (_Params[2] == "auto") {
			G_Driver_WarmUpTimeout = -1;
		} else {
			G_Driver_WarmUpTimeout = TL::ToInteger(_Params[2]);
		}
	} else if (_Method == C_Method_SetCanSpawn) {
		if (_Params.count < 2) return;
		declare Boolean[Text] Driver_SpawnPermissions for This = [];
		Driver_SpawnPermissions[_Params[0]] = (_Params[1] == "1");
	} else if (_Method == C_Method_EndMatch) {
		G_Driver_EndMatch = True;
	} else if (_Method == C_Method_SetLayer) {
		if (_Params.count < 2) return;
		declare Text Kind = "normal";
		if (_Params.count >= 3) Kind = _Params[2];
		Driver_SetLayer(_Params[0], _Params[1], Kind);
	} else if (_Method == C_Method_RemoveLayer) {
		if (_Params.count < 1) return;
		Driver_RemoveLayer(_Params[0]);
	} else if (_Method == C_Method_SetNetState) {
		if (_Params.count < 1) return;
		declare netwrite Text Net_Driver_State for Teams[0] = "";
		declare netwrite Integer Net_Driver_StateSerial for Teams[0] = 0;
		Net_Driver_State = _Params[0];
		Net_Driver_StateSerial += 1;
	} else if (_Method == C_Method_SetStandings) {
		if (_Params.count < 1) return;
		declare netwrite Text Net_Driver_Standings for Teams[0] = "";
		declare netwrite Integer Net_Driver_StandingsSerial for Teams[0] = 0;
		Net_Driver_Standings = _Params[0];
		Net_Driver_StandingsSerial += 1;
	} else if (_Method == C_Method_SetRound) {
		if (_Params.count < 1) return;
		declare netwrite Text Net_Driver_Round for Teams[0] = "";
		declare netwrite Integer Net_Driver_RoundSerial for Teams[0] = 0;
		Net_Driver_Round = _Params[0];
		Net_Driver_RoundSerial += 1;
	} else if (_Method == C_Method_SetUsers) {
		if (_Params.count < 1) return;
		declare netwrite Text Net_Driver_Users for Teams[0] = "";
		declare netwrite Integer Net_Driver_UsersSerial for Teams[0] = 0;
		Net_Driver_Users = _Params[0];
		Net_Driver_UsersSerial += 1;
	}
}

/// Applies the commands Match_Yield queued, in the order they arrived.
Void Driver_ApplyCommands() {
	declare Text[][] Driver_PendingCommands for This = [];
	foreach (Command in Driver_PendingCommands) {
		declare Text[] Params;
		foreach (Index => Param in Command) {
			if (Index > 0) Params.add(Param);
		}
		if (Command.count > 0) Driver_ApplyCommand(Command[0], Params);
	}
	Driver_PendingCommands.clear();
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/** Waits until the driver continues the hold.
 *	Commands keep being applied while waiting. The hold is sent again
 *	periodically because the driver answers repeated holds without
 *	deciding twice.
 *
 *	@param	_Section		Where the script waits (C_Section_*)
 *	@param	_Forced			RoundEnd only: the round should not count
 */
Void Driver_Hold(Text _Section, Boolean _Forced) {
	G_Driver_Seq += 1;
	declare K_Driver_Hold Hold = K_Driver_Hold {
		session = G_Driver_Session,
		seq = G_Driver_Seq,
		section = _Section,
		forced = _Forced
	};
	G_Driver_PendingSeq = Hold.seq;
	Driver_SendHold(Hold);

	declare Integer ResendTime = Now + S_DriverDecisionTimeout;
	declare Boolean Waiting = False;
	while (G_Driver_PendingSeq == Hold.seq && !ServerShutdownRequested) {
		MB_Yield();
		Driver_ApplyCommands();
		if (G_Driver_PendingSeq == Hold.seq && Now >= ResendTime) {
			if (!Waiting) {
				Waiting = True;
				UIModules_BigMessage::SetMessage(_("Waiting for the match driver..."));
			}
			Driver_SendHold(Hold);
			ResendTime = Now + C_Driver_ResendInterval;
		}
	}
	if (Waiting) UIModules_BigMessage::SetMessage("");
}

// ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~ //
/// Computes the end time of the round after the first player finished.
Integer Driver_FinishTimeout() {
	if (G_Driver_FinishTimeout == "off") return -1;
	if (G_Driver_FinishTimeout != "auto") return Now + TL::ToInteger(G_Driver_FinishTimeout);

	declare Integer FinishTimeout = 5000;
	if (Map.TMObjective_IsLapRace && Race::GetLapsNb() > 0 && Map.TMObjective_NbLaps > 0) {
		FinishTimeout += ((Map.TMObjective_AuthorTime / Map.TMObjective_NbLaps) * Race::GetLapsNb()) / 6;
	} else {
		FinishTimeout += Map.TMObjective_AuthorTime / 6;
	}
	return Now + FinishTimeout;
}
